private static Texture2D LoadPreviewTexture(BaseModPackage p, bool mipmap, out ModPreview.PreviewType kind, out bool owns) // [!code ++]
{ // [!code ++]
	kind = ModPreview.PreviewType.None; // [!code ++]
	owns = true; // [!code ++]
	FileInfo fileInfo = ModPreview.FindPreviewFile(p); // [!code ++]
	if (fileInfo != null) // [!code ++]
	{ // [!code ++]
		try // [!code ++]
		{ // [!code ++]
			Texture2D texture2D = ModPreview.LoadTexture(File.ReadAllBytes(fileInfo.FullName), out kind, mipmap); // [!code ++]
			if ((bool)texture2D) // [!code ++]
			{ // [!code ++]
				return texture2D; // [!code ++]
			} // [!code ++]
		} // [!code ++]
		catch (Exception ex) // [!code ++]
		{ // [!code ++]
			Debug.LogWarning("#mod preview " + fileInfo.FullName + ": " + ex.Message); // [!code ++]
		} // [!code ++]
	} // [!code ++]
	Texture2D texture2D2 = SteamPreview(p); // [!code ++]
	if ((bool)texture2D2) // [!code ++]
	{ // [!code ++]
		owns = false; // [!code ++]
		if (kind == ModPreview.PreviewType.None) // [!code ++]
		{ // [!code ++]
			kind = ModPreview.PreviewType.Jpeg; // [!code ++]
		} // [!code ++]
	} // [!code ++]
	return texture2D2; // [!code ++]
} // [!code ++]
